    generate2(_ps) {

        let c1 = color(random(255), random(255), random(255));
        let c2 = color(random(255), random(255), random(255));
        _ps.background(0);
        _ps.noStroke();
        _ps.rectMode(CENTER);
        _ps.fill(255);
        //bars();
        //bg();

        this.dibujarFondo(_ps);
        this.body(_ps);
        this.face(_ps);
    }
    body(_ps) {

        let cnt = 4;
        for (let i = 0; i < cnt; i++) {
            let rdm = 50;
            let y = height / 2 + random(-rdm, rdm);
            _ps.fill(random(255), 255);
            this.shape1(_ps,width / 2, y, random(300, 1500), random(200), random(PI / 3, -PI / 3), 0);
        }
    }
    face(_ps) {

        let cnt2 = 5;
        /*for(let i =0;i < cnt2; i++){
            let rdm = 70;
            let y = height/2 + random(-rdm,rdm);
            fill(random(255),random(255));
            shape1(width/2,y,random(1000),random(15));
            }*/

        this.dibujarCuerpo(_ps);

        if (random(1) < 0.4) {
            this.dibujarPelo(_ps);
        }

        _ps.ellipse(width / 2, height / 2, 200, 200);
        let cnt = 10;
        for (let i = 0; i < cnt; i++) {
            let rdm = 50;
            let y = height / 2 + random(-rdm, rdm);
            _ps.fill(random(255), random(170, 255));

            let w = random(800);
            let h = random(350);
            this.shape1(_ps,width / 2, y, w, h, random(-PI, PI) + PI / 2, 0);
            _ps.ellipse(width / 2, y, w, h);
        }

        this.dibujarOjos(_ps);
        this.dibujarBoca(_ps);
        this.dibujarNariz(_ps);

    }
    dibujarPelo(_ps) {
        let cnt3 = 800;
        for (let i = 0; i < cnt3; i++) {
            let s = map(i, 0, cnt3 - 1, 5, 0);

            let cf = lerpColor(this.c1, this.c2, random(1));
            let c = lerpColor(color(0), cf, map(i, 0, cnt3 - 1, 0, 1));
            //c.setAlpha(random(255));
            _ps.fill(c, random(255));
            let rdm = 300;
            this.shape1(_ps,width / 2 + random(-rdm, rdm),
                height / 2 + random(-rdm, rdm),
                random(350), s, random(TWO_PI * 4), 30);

            this.shape1(_ps,width / 2 + random(-rdm, rdm),
                height / 2 + random(-rdm, rdm),
                random(350), s, random(TWO_PI * 4), 30);
        }
    }
    dibujarOrejas(_ps) {
        let cnt3 = 2;
        for (let i = 0; i < cnt3; i++) {
            let s = map(i, 0, cnt3 - 1, 150, 0);

            let cf = lerpColor(this.c1, this.c2, random(1));
            let c = lerpColor(color(0), cf, map(i, 0, cnt3 - 1, 0, 1));
            _ps.fill(c);
            this.shape1(_ps,width / 2, height / 2 - 300, 120, 10, random(TWO_PI * 8), 30);
        }
    }
    dibujarNariz(_ps) {

            _ps.fill(c);
            this.shape1(_ps,width / 2, height / 2, s, s, random(PI / 2), 30);
        }
    }
    dibujarCuerpo(_ps) {
        let cnt3 = 20;
        for (let i = 0; i < cnt3; i++) {
            let s = map(i, 0, cnt3 - 1, 800, 0);
            let cf = lerpColor(this.c1, this.c2, random(1));
            let c = lerpColor(color(0), cf, map(i, 0, cnt3 - 1, 0, 1));
            fill(c, random(255));
            this.shape1(_ps,width / 2, height / 2, s, s, random(TWO_PI), 30);
            ellipse(width / 2, height / 2, s, s);
        }
    }
    dibujarOjos(_ps) {
        //OJOS 
        let cnt = floor(random(20));
        let sepx = random(20);
        let sepy = random(50, 150);

        let x = width / 2;
        let y = height / 2 - sepy;


        let h = random(5, 15);
        let an = random(100, 130);
        for (let i = 0; i < cnt; i++) {

            let w = random(50, 250);
            let ao = random(-PI, PI) + PI / 2;

            let cf = lerpColor(this.c1, color(0), random(1));
            fill(cf, random(100, 255));
            this.shape1(_ps,x - an, y, w, h, ao, sepx);
            this.shape1(_ps,x + an, y, w, h, ao + random(PI / 2), sepx);

            // ellipse(width/2-an, height/2-sepy, w+random(100), h+random(100));
            //ellipse(width/2+an, height/2-sepy, w+random(100), h+random(100));

        }
        let cnt2 = 10;

        for (let i = 0; i < cnt2; i++) {

            let idx = map(i, 0, cnt2 - 1, 0, 1);
            let w = map(i, 0, cnt2 - 1, random(150), 0);
            let hh = map(i, 0, cnt2 - 1, random(150), 0);
            let ao = random(-PI, PI) + PI / 2;

            let cf = lerpColor(this.c1, color(0), idx);
            _ps.fill(cf, random(100, 255));


            _ps.ellipse(x - an, y, w + random(100), hh);
            _ps.ellipse(x + an, y, w + random(100), hh);

        }
        _ps.fill(255, 255);
        _ps.ellipse(x - an, y, 5, 5);

        _ps.ellipse(x + an, y, 5, 5);
    }
    dibujarBoca(_ps) {
        //OJOS
        let cnt = 15;
        let size = random(200, 300);

        let cnt2 = 20;

        let x = width / 2;
        let y = height + size + random(50);
        for (let i = 0; i < cnt2; i++) {

            let idx = map(i, 0, cnt2 - 1, 0, 1);
            let w = map(i, 0, cnt2 - 1, random(300), 0);
            let hh = map(i, 0, cnt2 - 1, random(50), 0);
            let ao = random(-PI, PI) + PI / 2;

            let cf = lerpColor(this.c1, color(0), idx);
            _ps.fill(cf, 255);

            this.shape1(_ps,width / 2, height / 2 + size, w, 20, random(TWO_PI), 50);
        }


        for (let i = 0; i < cnt; i++) {
            let an = 150;
            let ao = random(-PI * 2, PI * 2) + PI / 2;
            ao = random(-PI * 4, PI * 4);
            let w = random(30, 150);
            w = map(i, 0, cnt - 1, random(120), 0);
            let h = random(50, 150);
            h = map(i, 0, cnt - 1, 20, 0);
            let sepx = random(60);
            let cf = lerpColor(this.c1, color(0), random(0, 1));
            cf = lerpColor(cf, color(255), random(0, 1));
            fill(cf, 255);
            this.shape1(_ps,x, height / 2 + size, w, h, ao, sepx);
        }


        for (let i = 0; i < cnt; i++) {
            let an = 150;
            let ao = random(-PI * 2, PI * 2) + PI / 2;
            ao = random(-PI * 4, PI * 4);
            let w = random(30, 120);
            w = map(i, 0, cnt - 1, random(500), random(100, 0));
            let h = random(50, 150);
            h = map(i, 0, cnt - 1, random(150), random(100));
            let sepx = random(60);
            let cf = lerpColor(this.c1, color(0), random(0, 1));
            cf = lerpColor(cf, color(255), random(0, 1));
            _ps.fill(cf, random(255));
            _ps.ellipse(x, height / 2 + size, w, h);
        }


    }
    shape1(_ps,x,
        y,
        w,
        h,
        a,
        sepx) {

        // let w =random(800);
        //  let h =random(100);
        let ar = a;
        _ps.push();
        _ps.translate(x, y);
        _ps.rotate(ar);
        _ps.rect(0 + sepx, 0, w, h);
        _ps.pop();

        _ps.push();
        _ps.translate(x, y);
        _ps.rotate(-ar);
        _ps.rect(0 - sepx, 0, w, h);
        _ps.pop();
    }

    bg(_ps) {
        let cnt = 5;
        let s = 20;
        for (let i = 0; i < cnt; i++) {
            let idx = map(i, 0, cnt - 1, 0, 1);

                width * 1 / 8,

            _ps.rect(x, height / 2, s, height);

    bars(_ps) {

    }

    dibujarFondo(_ps) {

        let cnt3 = 30;
        for (let i = 0; i < cnt3; i++) {
            let s = map(i, 0, cnt3 - 1, width * 1.4, 0);
            let idx = map(i, 0, cnt3 - 1, 0, 1);

            let c3 = lerpColor(this.c1, this.c2, random(1));
            let cf = lerpColor(c3, color(0), sin(idx * 50 + random(TWO_PI)) * .5 + .5);
            cf = lerpColor(cf, color(255), 0.2);
            cf = lerpColor(cf, color(0), 0.2);
            //cf = lerpColor(cf, color(255), random(0, 1));
            _ps.fill(cf, 150);
            _ps.ellipse(width / 2, height / 2, s, s);
        }
    }
