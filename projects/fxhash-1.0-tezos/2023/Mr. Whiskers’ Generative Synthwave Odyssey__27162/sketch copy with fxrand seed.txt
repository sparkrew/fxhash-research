function createRandomNumbers(seed, count = 10000) {
  let randomNumbers = [];
  fxrand.setSeed(seed); // Presuming fxrand has a method to set the seed

  for (let i = 0; i < count; i++) {
    randomNumbers.push(fxrand());
  }

// Initialize music components
let randomNumbers = [];
let randomNumberIndex = 0;

function initMusic(customPitches, seed) {
  randomNumberIndex = 0;
  randomNumbers = generateRandomNumbers(seed);

  if (customPitches) {
    pitches = customPitches;
  } else {
    pitches = pitchSets[Math.floor(randomNumbers[randomNumberIndex++] * pitchSets.length)];
  }
  bpm = Math.floor(randomNumbers[randomNumberIndex++] * (180 - 120)) + 120;

  const loopDuration = 60 / bpm * 0.5; // 8th note
  const cells = Array(pitches.length * numTimeSteps).fill(false);

  synth = new p5.PolySynth();
  const reverb = new p5.Reverb();

  pokem = setInterval(() => {
    // Change the scale at random intervals
    if (randomNumbers[randomNumberIndex++] < 0.1) {
      const newPitches = pitchSets[Math.floor(randomNumbers[randomNumberIndex++] * pitchSets.length)];
      for (let i = 0; i < pitches.length; i++) {
        pitches[i] = newPitches[i];
      }
    }
  }, loopDuration * 1000);

  reverb.process(synth, 3, 2);

  soundLoop = new p5.SoundLoop((cycleStartTime) => {
    // Randomly enable or disable cells in every loop
    for (let i = 0; i < cells.length; i++) {
      cells[i] = randomNumbers[randomNumberIndex++] < 0.2; // 20% chance for a cell to be enabled
    }

    for (let i = 0; i < cells.length; i++) {
      if (Math.floor(i / pitches.length) === timeStepCounter) {
        if (cells[i]) {
          // Play sound
          const velocity = 1; // Between 0-1
          const quaverSeconds = soundLoop._convertNotation('16n'); // 8th note = quaver duration
          const freq = midiToFreq(pitches[i % pitches.length]);
          synth.play(freq, velocity, cycleStartTime, quaverSeconds);
        }
      }
    }
    soundLoop.bpm = bpm;
    timeStepCounter = (timeStepCounter + 1) % numTimeSteps;
  }, '8n');

  soundLoop.bpm = bpm;
}
function generateRandomNumbers(seed) {
  let randomNumbers = Array(10000).fill().map(() => fxrand(seed));
  return randomNumbers;
}


function midiToFreq(midiNote) {
  return 440 * Math.pow(2, (midiNote - 69) / 12);
}

function releaseAllNotes() {
  if (synth) {
    for (const note of pitches) {
      synth.noteRelease(midiToFreq(note));
    }
  }
}


// Listen for key press events
window.addEventListener('keydown', (event) => {
  if (event.key === 'z' || event.key === 'Z') {
  // Resume the audio context in the same user gesture
    if (p5.soundOut.audiocontext.state === 'suspended') {
      p5.soundOut.audiocontext.resume();
    }
    if (!soundLoop.isPlaying) {
      soundLoop.start();
    } else {
      soundLoop.stop();
    }
  }
});

// This function will start the music
function startMusic(seed) {
  // Resume the AudioContext
  if (p5.soundOut.audiocontext.state === 'suspended') {
    p5.soundOut.audiocontext.resume();
  }
  
  // Initialize the music
  initMusic(null, seed);
  
  // Start the soundLoop
  soundLoop.start();
  
  // Remove the event listeners to prevent the music from restarting if the canvas is clicked again or 'm' is pressed again
  canvas.removeEventListener('click', startMusic);
  canvas.removeEventListener('touchstart', startMusic);
  window.removeEventListener('keydown', startMusicOnM);
}

// This function will start the music if 'm' is pressed
function startMusicOnM(event) {
  if (event.key === 'm') {
    startMusic(seed);
  }
}
